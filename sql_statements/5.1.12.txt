-- Run with psql -v ON_ERROR_STOP=1 so a failure stops the migration.
-- Do not pass --single-transaction: CREATE INDEX CONCURRENTLY cannot run inside a
-- transaction block. A failed concurrent build leaves an INVALID index to drop before
-- rerunning. Every statement is idempotent.
--
-- Apply before deploying the application; db.create_all() does not add a column to an
-- existing table. Do not drop anything here while an instance might still run the new code.
--
-- Shared-key activity summaries filter requests by their funding key. Build the lookup index
-- without blocking ordinary generation writes, before deploying the activity endpoint.
CREATE INDEX CONCURRENTLY IF NOT EXISTS ix_waiting_prompts_sharedkey_id
  ON waiting_prompts (sharedkey_id)
  WHERE sharedkey_id IS NOT NULL;

-- No default, so the add is metadata-only and existing rows stay unknown. Provenance is
-- never backfilled from the effective prompt. The lock timeout fails rather than queueing
-- an ACCESS EXCLUSIVE lock behind a long transaction.
SET lock_timeout = '5s';
ALTER TABLE waiting_prompts ADD COLUMN IF NOT EXISTS submitted_prompt TEXT;
RESET lock_timeout;

-- No account or request foreign keys: a rejected submission has no waiting row, and
-- evidence must outlive request expiry and account deletion.
CREATE TABLE IF NOT EXISTS prompt_moderation_events (
  id BIGSERIAL PRIMARY KEY,
  created TIMESTAMP WITHOUT TIME ZONE NOT NULL DEFAULT now(),
  user_id INTEGER NOT NULL,
  request_id VARCHAR(36),
  job_id VARCHAR(36) UNIQUE,
  worker_id VARCHAR(36),
  proxied_account TEXT,
  reason VARCHAR(32) NOT NULL,
  outcome VARCHAR(16) NOT NULL,
  submitted_prompt TEXT,
  moderation_prompt TEXT,
  effective_prompt TEXT,
  text_truncated BOOLEAN NOT NULL DEFAULT false
);
CREATE INDEX IF NOT EXISTS ix_prompt_moderation_created ON prompt_moderation_events (created);
CREATE INDEX IF NOT EXISTS ix_prompt_moderation_user_id ON prompt_moderation_events (user_id, id);
CREATE INDEX IF NOT EXISTS ix_prompt_moderation_reason_id ON prompt_moderation_events (reason, id);

-- Review disposition is separate so that recording it never edits captured evidence.
CREATE TABLE IF NOT EXISTS prompt_moderation_reviews (
  event_id BIGINT PRIMARY KEY REFERENCES prompt_moderation_events (id) ON DELETE CASCADE,
  status VARCHAR(16) NOT NULL,
  reviewer_id INTEGER NOT NULL,
  reviewed_at TIMESTAMP WITHOUT TIME ZONE NOT NULL DEFAULT now(),
  note VARCHAR(2000) NOT NULL DEFAULT ''
);

-- worker_suspicions remains the resettable current score. No foreign keys here: deleting a
-- worker or resetting its suspicion must not delete the history moderators review.
CREATE TABLE IF NOT EXISTS worker_suspicion_events (
  id BIGSERIAL PRIMARY KEY,
  created TIMESTAMP WITHOUT TIME ZONE NOT NULL DEFAULT now(),
  worker_id UUID NOT NULL,
  worker_name VARCHAR(100) NOT NULL,
  user_id INTEGER NOT NULL,
  suspicion_id INTEGER NOT NULL,
  amount INTEGER NOT NULL DEFAULT 1,
  detail TEXT NOT NULL
);
CREATE INDEX IF NOT EXISTS ix_worker_suspicion_events_created
  ON worker_suspicion_events (created);
CREATE INDEX IF NOT EXISTS ix_worker_suspicion_events_worker_id
  ON worker_suspicion_events (worker_id, id);
CREATE INDEX IF NOT EXISTS ix_worker_suspicion_events_user_id
  ON worker_suspicion_events (user_id, id);
CREATE INDEX IF NOT EXISTS ix_worker_suspicion_events_reason_id
  ON worker_suspicion_events (suspicion_id, id);
